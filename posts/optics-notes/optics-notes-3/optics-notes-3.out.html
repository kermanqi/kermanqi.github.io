Wave packets

This notebook illustrates a simple harmonic wave and the wave group obtained by combining two nearby waves. The carrier, envelope, and wave group are shown in separate panels. Use the play button or the time slider to inspect their evolution.

For two nearby waves, the wave group can be written schematically as

\[V_{\mathrm{group}}(z,t) = 2a\cos\left[ rac{1}{2}(\Delta\omega\,t-\Delta k\,z) ight]\cos(ar\omega t-ar k z).\]

Parameters and wave functions

The first code block imports the plotting libraries, defines the spatial grid and wave parameters, and provides a reusable function for evaluating the carrier, envelope, and wave group at any time.

import numpy as np
import plotly.graph_objects as go
from plotly.subplots import make_subplots


# --------------------------------------------------
# Parameters
# --------------------------------------------------

a = 1.0

omega_bar = 8.0       # central angular frequency
k_bar = 8.0           # central wave number

delta_omega = 0.8     # frequency difference
delta_k = 0.8         # wave-number difference

z = np.linspace(0, 8 * np.pi, 2500)


def calculate_waves(t):
    """Calculate carrier, envelope and wave group at time t."""

    phase = omega_bar * t - k_bar * z

    envelope_phase = 0.5 * (
        delta_omega * t - delta_k * z
    )

    carrier = a * np.cos(phase)

    envelope = 2 * a * np.cos(envelope_phase)

    wave_group = envelope * np.cos(phase)

    return carrier, envelope, wave_group

Initial state

The next block creates the two-panel figure and adds the simple harmonic wave, the wave group, and its positive and negative envelopes.

# --------------------------------------------------
# Initial state
# --------------------------------------------------

carrier, envelope, wave_group = calculate_waves(t=0)


fig = make_subplots(
    rows=2,
    cols=1,
    shared_xaxes=True,
    vertical_spacing=0.12,
    row_heights=[0.38, 0.62]
)


# (a) Simple harmonic wave

fig.add_trace(
    go.Scatter(
        x=z,
        y=carrier,
        mode="lines",
        name="simple wave",
        hovertemplate="z = %{x:.3f}<br>V = %{y:.3f}<extra></extra>"
    ),
    row=1,
    col=1
)


# (c) Wave group

fig.add_trace(
    go.Scatter(
        x=z,
        y=wave_group,
        mode="lines",
        name="wave group",
        hovertemplate="z = %{x:.3f}<br>V = %{y:.3f}<extra></extra>"
    ),
    row=2,
    col=1
)


# (b) Positive envelope

fig.add_trace(
    go.Scatter(
        x=z,
        y=envelope,
        mode="lines",
        name="envelope",
        line=dict(dash="dash"),
        hovertemplate="z = %{x:.3f}<br>Envelope = %{y:.3f}<extra></extra>"
    ),
    row=2,
    col=1
)


# Negative envelope

fig.add_trace(
    go.Scatter(
        x=z,
        y=-envelope,
        mode="lines",
        line=dict(dash="dash"),
        showlegend=False,
        hoverinfo="skip"
    ),
    row=2,
    col=1
)

Animation frames

Each frame evaluates the same wave functions at a different time. The four traces are updated together so the envelope remains synchronized with the wave group.

# --------------------------------------------------
# Animation frames
# --------------------------------------------------

t_values = np.linspace(
    0,
    2 * np.pi / delta_omega,
    61
)

frames = []

for t in t_values:

    carrier_t, envelope_t, group_t = calculate_waves(t)

    frames.append(
        go.Frame(
            name=f"{t:.3f}",
            data=[
                go.Scatter(y=carrier_t),
                go.Scatter(y=group_t),
                go.Scatter(y=envelope_t),
                go.Scatter(y=-envelope_t),
            ],
            traces=[0, 1, 2, 3]
        )
    )

fig.frames = frames

Playback controls

The layout contains a play/pause control and a time slider. The slider labels are shown at regular intervals to keep the axis readable.

# --------------------------------------------------
# Play button and time slider
# --------------------------------------------------

fig.update_layout(

    template="simple_white",

    width=900,
    height=560,

    margin=dict(
        l=60,
        r=25,
        t=35,
        b=85

    hovermode="x",

    legend=dict(
        orientation="h",
        x=0.5,
        xanchor="center",
        y=1.06,
        yanchor="bottom"

    updatemenus=[
        dict(
            type="buttons",
            direction="left",
            x=0.0,
            y=-0.14,

            buttons=[
                dict(
                    label="▶ Play",
                    method="animate",
                    args=[
                        None,
                        {
                            "frame": {
                                "duration": 45,
                                "redraw": False

                                "duration": 0

                dict(

                            "mode": "immediate",

                            },
                            "transition": {
                                "duration": 0

                            }
                        }
                    ]
                )

                for i, t in enumerate(t_values)

        )
    ]
